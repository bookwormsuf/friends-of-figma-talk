import test from 'node:test';
import assert from 'node:assert/strict';
import {advanceClosing,closingConfig as C,closingState,type ClosingSize,type ClosingVariation} from '../src/slides/closingMotion.ts';
const variations=[1,2,3] as ClosingVariation[],sizes=[1,2,3] as ClosingSize[];
/** Distance from the stage centre in pixels on the 1280 by 720 stage, so x and y compare fairly. */
const fromCentre=(o:{x:number;y:number})=>Math.hypot((o.x-50)*12.8,(o.y-50)*7.2);
for(const v of variations)for(const size of sizes){
 const name=`ending ${v} size ${size}`;
 test(`${name}: starts with the words alone`,()=>{
  const s=closingState(v,size,0);
  assert.equal(s.words,0);assert.equal(s.others.length,C.others);
  assert.ok(s.others.every(o=>!o.visible),'no cloud is on stage yet');assert.equal(s.centre.visible,false);
  assert.equal(s.glow.cloud,0);assert.equal(s.glow.spark,0);assert.equal(s.tint,0);
 });
 test(`${name}: the centre cloud is hidden inside the swarm before it parts`,()=>{
  // Just before the others move out: every cloud has arrived, and the centre one is still small with clouds on top of it.
  const s=closingState(v,size,.45),small=C.sizes[size].small;
  assert.ok(s.others.every(o=>o.visible));
  assert.ok(Math.abs(s.centre.width-small)<1e-9,'same size as the others');
  const reach=small*12.8*.6;
  assert.ok(s.others.some(o=>fromCentre(o)<reach),'at least one cloud overlaps the centre');
  assert.equal(s.glow.cloud,0);
 });
 test(`${name}: press one ends on a glowing cloud inside a ring`,()=>{
  const s=closingState(v,size,1),swarm=closingState(v,size,.45);
  assert.equal(s.words,1);
  const swarmRadius=Math.max(...swarm.others.map(fromCentre));
  for(const o of s.others){
   assert.ok(fromCentre(o)>swarmRadius,'each ring cloud is farther out than the swarm ever was');
   const half=C.sizes[size].small/2;
   assert.ok(o.x-half>0&&o.x+half<100&&o.y>4&&o.y<96,`ring cloud stays on stage (${o.x.toFixed(1)}, ${o.y.toFixed(1)})`);
  }
  assert.deepEqual([s.centre.x,s.centre.y],[50,50]);
  assert.ok(Math.abs(s.centre.width-C.sizes[size].centre)<1e-9);
  assert.equal(s.centre.reshape,0);assert.equal(s.centre.rotation,0);
  assert.ok(Math.abs(s.glow.cloud-1)<1e-9);assert.equal(s.glow.spark,0);assert.equal(s.tint,0);
  assert.ok(Math.abs(s.glow.spread-C.spread.cloud)<1e-9);
 });
 test(`${name}: press two ends on a glowing spark the same width as the cloud`,()=>{
  const one=closingState(v,size,1),s=closingState(v,size,2);
  assert.equal(s.centre.reshape,1);assert.equal(s.centre.stagger,0);assert.ok(Math.abs(s.centre.rotation)<1e-9);
  assert.ok(Math.abs(s.centre.width-one.centre.width)<1e-9);
  assert.ok(Math.abs(s.glow.spark-1)<1e-9);assert.ok(Math.abs(s.glow.cloud)<1e-9);assert.ok(Math.abs(s.tint-1)<1e-9);
  assert.ok(s.glow.spread>one.glow.spread,'the spark glows wider than the cloud did');
  // The ring does not move during press two.
  s.others.forEach((o,i)=>{assert.equal(o.x,one.others[i].x);assert.equal(o.y,one.others[i].y)});
 });
 test(`${name}: nothing jumps between frames`,()=>{
  let before=closingState(v,size,0);
  for(let p=.002;p<=2;p+=.002){
   const now=closingState(v,size,p);
   now.others.forEach((o,i)=>{if(o.visible&&before.others[i].visible)assert.ok(fromCentre({x:50+o.x-before.others[i].x,y:50+o.y-before.others[i].y})<40,`cloud ${i} jumps at p=${p.toFixed(3)}`)});
   assert.ok(Math.abs(now.centre.reshape-before.centre.reshape)<.03,`reshape jumps at p=${p.toFixed(3)}`);
   assert.ok(Math.abs(now.glow.spread-before.glow.spread)<.05,`glow jumps at p=${p.toFixed(3)}`);
   before=now;
  }
 });
}
for(const v of variations){
 test(`ending ${v}: playing forward takes the pinned seconds for each press, and back returns to the start`,()=>{
  const [one,two]=C.seconds[v],frame=1/60;
  let p=0,t=0;while(p<1&&t<60){p=advanceClosing(p,1,frame,v);t+=frame}
  assert.equal(p,1);assert.ok(Math.abs(t-one)<.1,`press one took ${t.toFixed(2)}s`);
  t=0;while(p<2&&t<60){p=advanceClosing(p,2,frame,v);t+=frame}
  assert.equal(p,2);assert.ok(Math.abs(t-two)<.1,`press two took ${t.toFixed(2)}s`);
  t=0;while(p>1&&t<60){p=advanceClosing(p,1,frame,v);t+=frame}
  assert.equal(p,1);assert.ok(Math.abs(t-two)<.1);
  t=0;while(p>0&&t<60){p=advanceClosing(p,0,frame,v);t+=frame}
  assert.equal(p,0);assert.ok(Math.abs(t-one)<.1);
 });
 test(`ending ${v}: a second press before the first has finished hurries on instead of waiting`,()=>{
  const [one,two]=C.seconds[v],frame=1/60;
  let p=0,t=0;while(p<2&&t<60){p=advanceClosing(p,2,frame,v);t+=frame}
  assert.equal(p,2);assert.ok(t<one+two,'faster than the two presses played in full');
 });
 test(`ending ${v}: a stalled or broken frame time never throws p out of range`,()=>{
  for(const delta of [NaN,-1,Infinity,10]){const p=advanceClosing(.5,2,delta,v);assert.ok(p>=.5&&p<=2&&Number.isFinite(p))}
  assert.equal(advanceClosing(1,1,.016,v),1);
 });
}
